const SIZE = 8;

const comboElement = document.getElementById("combo");


let board;

let pieces;


let score = 0;


let highscore =

    Number(localStorage.getItem("blockblast_highscore")) || 0;


let dragging = null;


const colors = [

    "#ff7675",

    "#74b9ff",

    "#55efc4",

    "#ffeaa7",

     [0,0,1]]

];



// ============================

// KHỞI TẠO GAME

// ============================


function restart() {


    board = [];


    for(let r = 0; r < SIZE; r++) {


        board[r] = [];


        for(let c = 0; c < SIZE; c++) {

            board[r][c] = null;

        }

    }


    score = 0;


    gameoverElement.classList.remove("show");


    updateScore();


    createPieces();


    renderBoard();

}



// ============================

// TẠO 3 KHỐI

// ============================


function createPieces() {


    pieces = [];


    for(let i = 0; i < 3; i++) {


        const shape =

            SHAPES[

                Math.floor(Math.random() * SHAPES.length)

            ];


        pieces.push({

            shape: shape,

            color:

                colors[

                    Math.floor(Math.random() * colors.length)

                ],

            used: false

        });

    }


    renderPieces();

}



// ============================

// VẼ BÀN CỜ

// ============================


function renderBoard() {


    boardElement.innerHTML = "";


    for(let r = 0; r < SIZE; r++) {


        for(let c = 0; c < SIZE; c++) {


            const cell = document.createElement("div");


            cell.className = "cell";


            cell.dataset.row = r;

            cell.dataset.col = c;


            if(board[r][c]) {


                cell.classList.add("filled");


                cell.style.background =

                    board[r][c];

            }


            boardElement.appendChild(cell);

        }

    }

}



// ============================

// VẼ KHỐI BÊN DƯỚI

// ============================


function renderPieces() {


    piecesElement.innerHTML = "";


    pieces.forEach((piece,index) => {


        const container =

            document.createElement("div");


        container.className = "piece";


        if(piece.used) {

            container.style.visibility = "hidden";

        }


        const shape =

            document.createElement("div");


        shape.className = "shape";


        shape.style.gridTemplateColumns =

            `repeat(${piece.shape[0].length}, 24px)`;


        piece.shape.forEach(row => {


            row.forEach(value => {


                const block =

                    document.createElement("div");


                if(value) {


                    block.className = "block";


                    block.style.background =

                        piece.color;

                }


                shape.appendChild(block);

            });

        });


        container.appendChild(shape);


        addDragEvents(container,index);


        piecesElement.appendChild(container);

    });

}



// ============================

// KÉO KHỐI

// ============================


function addDragEvents(element,index) {


    element.addEventListener(

        "pointerdown",

        e => {


            if(pieces[index].used)

                return;


            dragging = {

                index:index,

                x:e.clientX,

                y:e.clientY

            };


            element.setPointerCapture(e.pointerId);


            element.style.transform =

                "scale(1.15)";


            e.preventDefault();

        }

    );


    element.addEventListener(

        "pointermove",

        e => {


            if(!dragging ||

               dragging.index !== index)

                return;


            const rect =

                boardElement.getBoundingClientRect();


            const cellWidth =

                rect.width / SIZE;


            const col =

                Math.floor(

                    (e.clientX - rect.left)

                    / cellWidth

                );


            const row =

                Math.floor(

                    (e.clientY - rect.top)

                    / cellWidth

                );


            clearPreview();


            if(canPlace(

                pieces[index].shape,

                row,

                col

            )) {


                preview(

                    pieces[index],

                    row,

                    col

                );

            }

        }

    );


    element.addEventListener(

        "pointerup",

        e => {


            if(!dragging ||

               dragging.index !== index)

                return;


            const rect =

                boardElement.getBoundingClientRect();


            const cellWidth =

                rect.width / SIZE;


            const col =

                Math.floor(

                    (e.clientX - rect.left)

                    / cellWidth

                );


            const row =

                Math.floor(

                    (e.clientY - rect.top)

                    / cellWidth

                );


            element.style.transform = "";


            clearPreview();


            if(canPlace(

                pieces[index].shape,

                row,

                col

            )) {


                placePiece(

                    pieces[index],

                    row,

                    col

                );


                pieces[index].used = true;


                renderPieces();


                if(

                    pieces.every(

                        piece => piece.used

                    )

                ) {

                    createPieces();

                }


                setTimeout(checkGameOver,150);

            }


            dragging = null;

        }

    );

}



// ============================

// KIỂM TRA CÓ ĐẶT ĐƯỢC KHÔNG

// ============================


function canPlace(shape,row,col) {


    for(let r = 0; r < shape.length; r++) {


        for(let c = 0; c < shape[r].length; c++) {


            if(!shape[r][c])

                continue;


            const rr = row+r;

            const cc = col+c;


            if(

                rr < 0 ||

                rr >= SIZE ||

                cc < 0 ||

                cc >= SIZE

            ) {

                return false;

            }


            if(board[rr][cc]) {

                return false;

            }

        }

    }


    return true;

}



// ============================

// ĐẶT KHỐI

// ============================


function placePiece(piece,row,col) {


    let cells = 0;


    for(let r = 0; r < piece.shape.length; r++) {


        for(let c = 0; c < piece.shape[r].length; c++) {


            if(piece.shape[r][c]) {


                board[row+r][col+c] =

                    piece.color;


                cells++;

            }

        }

    }


    score += cells * 5;


    renderBoard();


    setTimeout(clearLines,100);


    updateScore();

}



// ============================

// XÓA HÀNG / CỘT

// ============================


function clearLines() {


    let rows = [];

    let cols = [];


    // Hàng


    for(let r = 0; r < SIZE; r++) {


        if(

            board[r].every(

                cell => cell !== null

            )

        ) {

            rows.push(r);

        }

    }


    // Cột


    for(let c = 0; c < SIZE; c++) {


        let full = true;


        for(let r = 0; r < SIZE; r++) {


            if(!board[r][c]) {


                full = false;

                break;

            }

        }


        if(full) {

            cols.push(c);

        }

    }


    const total =

        rows.length + cols.length;


    if(total === 0)

        return;


    // Xóa hàng


    rows.forEach(r => {


        for(let c = 0; c < SIZE; c++) {

            board[r][c] = null;

        }

    });


    // Xóa cột


    cols.forEach(c => {


        for(let r = 0; r < SIZE; r++) {

            board[r][c] = null;

        }

    });


    /*

        Tính điểm:

        1 line = 10

        2 line = 30

        3 line = 60

        4 line = 100

    */


    score +=

        total * 10 +

        (total > 1

            ? total * total * 5

            : 0);


    showCombo(total);


    renderBoard();


    updateScore();

}



// ============================

// PREVIEW

// ============================


function preview(piece,row,col) {


    for(let r = 0; r < piece.shape.length; r++) {


        for(let c = 0; c < piece.shape[r].length; c++) {


            if(!piece.shape[r][c])

                continue;


            const rr = row+r;

            const cc = col+c;


            if(

                rr < 0 ||

                rr >= SIZE ||

                cc < 0 ||

                cc >= SIZE

            )

                continue;


            const index =

                rr * SIZE + cc;


            const cell =

                boardElement.children[index];


            cell.classList.add("preview");


            cell.style.background =

                piece.color;

        }

    }

}



function clearPreview() {


    document

        .querySelectorAll(".preview")

        .forEach(cell => {


            cell.classList.remove("preview");


            const r =

                Number(cell.dataset.row);


            const c =

                Number(cell.dataset.col);


            cell.style.background =

                board[r][c] || "";

        });

}



// ============================

// GAME OVER

// ============================


function checkGameOver() {


    for(const piece of pieces) {


        if(piece.used)

            continue;


        for(let r = 0; r < SIZE; r++) {


            for(let c = 0; c < SIZE; c++) {


                if(

                    canPlace(

                        piece.shape,

                        r,

                        c

                    )

                ) {

                    return;

                }

            }

        }

    }


    endGame();

}



function endGame() {


    if(score > highscore) {


        highscore = score;


        localStorage.setItem(

            "blockblast_highscore",

            highscore

        );

    }


    finalScoreElement.textContent =

        score;


    finalHighElement.textContent =

        highscore;


    gameoverElement.classList.add(

        "show"

    );


    updateScore();

}



// ============================

// ĐIỂM

// ============================


function updateScore() {


    scoreElement.textContent =

        score;


    highscoreElement.textContent =

        highscore;

}



// ============================

// COMBO

// ============================


function showCombo(lines) {


    if(lines <= 1)

        return;


    comboElement.textContent =

        lines === 2

            ? "🔥 DOUBLE!"

            : lines === 3

                ? "🔥 TRIPLE!"

                : "💥 AMAZING!";


    comboElement.classList.remove(

        "show"

    );


    void comboElement.offsetWidth;


    comboElement.classList.add(

        "show"

    );

}



// ============================

// START

// ============================


restart();
